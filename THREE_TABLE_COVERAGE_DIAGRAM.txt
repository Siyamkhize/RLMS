SQL QUERY STRUCTURE:
════════════════════

CREATE TEMPORARY TABLE temp_learner_coverage AS
SELECT 
    learnerID,
    COUNT(DISTINCT unit_standard_source) as total_unit_standards
FROM (
    -- Source 1: POE documents
    SELECT DISTINCT 
        learnerID, 
        'poe' as unit_standard_source
    FROM poe 
    WHERE filePath IS NOT NULL AND filePath != ''
    
    UNION
    
    -- Source 2: Assessment marks
    SELECT DISTINCT 
        learnerID, 
        CONCAT('marks_', unit_standard_id) as unit_standard_source
    FROM marks
    WHERE marks_scored IS NOT NULL
    
    UNION
    
    -- Source 3: Logbook marks
    SELECT DISTINCT 
        learnerID, 
        CONCAT('logbook_', unit_standard_id) as unit_standard_source
    FROM logbook_marks
    WHERE marks IS NOT NULL
) AS all_coverage
GROUP BY learnerID;
