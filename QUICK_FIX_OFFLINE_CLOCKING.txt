STEP 2: Update clock_in_page.dart
────────────────────────────────────────────────────────────────
@override
void initState() {
  super.initState();
  _checkLocalDataAvailability(); // ADD THIS LINE
  _initializeData();
  // ... rest of init
}

// ADD THIS METHOD:
Future<void> _checkLocalDataAvailability() async {
  final hasData = await DatabaseHelper().hasLocalLearnerData(widget.classID);
  if (!hasData) {
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text('No local data. Please sync when online.')),
    );
  }
}

STEP 3: Update sync_service.dart
────────────────────────────────────────────────────────────────
In _syncLearnerDetails() method:

// REMOVE THIS LINE:

for (var learner in learners) {
  await _dbHelper.upsertLearner(learner); // Use upsert
